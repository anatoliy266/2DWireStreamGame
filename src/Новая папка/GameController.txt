import { PlayerNode } from "./classes/PlayerNode";
import { Enemy } from "./classes/Enemy";
import { NetworkBuffer } from "./classes/NetworkBuffer";
import { DungeonMaster } from "./classes/DungeonMaster";

export class GameController {
    players: Map<string, PlayerNode> = new Map();
    enemies: Enemy[] = [];
    buffer: NetworkBuffer;
    dungeon: DungeonMaster;
    
    isGameRunning: boolean = false;
    turnTimer: any = null;
    timeLeft: number = 0;
    readonly TURN_DURATION = 15;

    constructor() {
        this.buffer = new NetworkBuffer();
        this.dungeon = new DungeonMaster();
    }

    startGame() {
        this.isGameRunning = true;
        console.log("=== INFILTRATION STARTED ===");
        this.nextLevel();
    }

    private nextLevel() {
        this.enemies = this.dungeon.loadNextHop();
        console.log(`--- ENTERING HOP ${this.dungeon.currentHop} ---`);
        
        if (this.enemies.length === 0 && this.dungeon.isVictory()) {
            console.log("SYSTEM BREACHED. MISSION ACCOMPLISHED.");
            this.isGameRunning = false;
            return;
        }

        // Охлаждение перед новым уровнем (-50ms)
        this.players.forEach(p => {
            p.latency = Math.max(0, p.latency - 50);
            if(p.status === 'BLACKOUT') p.status = 'OPERATIONAL';
        });

        this.startTurn();
    }

    private startTurn() {
        this.timeLeft = this.TURN_DURATION;
        this.buffer.reset();

        // 1. Привязка врагов к слотам атаки
        this.enemies.forEach(e => {
            // Босс занимает все слоты, мобы - свои конкретные
            if (e.type === 'MAINFRAME') {
                [0, 1, 2].forEach(i => this.buffer.attackSlots[i].assignedEntityId = e.id);
            } else {
                if (this.buffer.attackSlots[e.linkedSlotIndex]) {
                    this.buffer.attackSlots[e.linkedSlotIndex].assignedEntityId = e.id;
                }
            }
        });

        // 2. Назначение игроков в защиту (Designated Nodes)
        const activePlayers = Array.from(this.players.values())
            .filter(p => p.status !== 'TERMINATED');
        
        // Случайный выбор 4-х игроков
        const shuffled = activePlayers.sort(() => 0.5 - Math.random()).slice(0, 4);
        
        this.buffer.defenseSlots.forEach((slot, index) => {
            if (shuffled[index]) {
                slot.assignedEntityId = shuffled[index].id;
                console.log(`[DEF] Slot D${index+1} assigned to ${shuffled[index].name}`);
            }
        });

        console.log(`--- TURN START (${this.timeLeft}s) ---`);
        
        // Таймер
        clearInterval(this.turnTimer);
        this.turnTimer = setInterval(() => {
            this.timeLeft--;
            if (this.timeLeft <= 0) {
                this.resolveTurn();
            }
        }, 1000);
    }

    // === ОБРАБОТКА КОМАНД (Twitch -> Logic) ===
    handleInput(userId: string, userName: string, command: string, args: any) {
        if (!this.isGameRunning) return;

        let player = this.players.get(userId);
        if (!player) {
            player = new PlayerNode(userId, userName);
            this.players.set(userId, player);
        }

        // Проверки статуса
        if (player.status === 'TERMINATED' || player.status === 'BLACKOUT') return;

        // Расчет стоимости (Cost)
        let cost = 10;
        if (command.startsWith('sudo')) cost = 30;
        if (command.startsWith('ping')) cost = 5;

        player.addLatency(cost);

        // Логика Буфера
        let success = false;
        
        if (command.includes('attack')) {
            success = this.buffer.addAttackToken();
        } 
        else if (command.includes('defend')) {
            success = this.buffer.addDefenseToken(this.players);
        }
        else if (command.includes('sys_lock')) {
             // Спец команда (ульта)
             player.addLatency(100);
             // Реализация ульты: полное закрытие слота защиты...
             success = true; 
        }

        // OVERFLOW Check
        if (!success) {
            console.log(`OVERFLOW triggered by ${player.name}!`);
            // Штраф всему рейду
            this.players.forEach(p => p.addLatency(40));
        }
    }

    private resolveTurn() {
        clearInterval(this.turnTimer);
        console.log("--- EXECUTING BUFFER ---");

        // 1. Игроки атакуют врагов
        this.buffer.attackSlots.forEach(slot => {
            if (slot.count > 0 && slot.assignedEntityId) {
                // Если босс, находим его по ID (он один), если моб - по ID
                const enemy = this.enemies.find(e => e.id === slot.assignedEntityId);
                if (enemy) {
                    const dmg = slot.count * 20; // Base Damage
                    enemy.takeDamage(dmg);
                    console.log(`> ATTACK A${slot.index+1} -> ${enemy.name}: -${dmg} HP`);
                }
            }
        });

        // Удаляем мертвых
        this.enemies = this.enemies.filter(e => !e.isDead);

        if (this.enemies.length === 0) {
            this.nextLevel();
            return;
        }

        // 2. Враги атакуют игроков
        this.buffer.defenseSlots.forEach(slot => {
            if (slot.assignedEntityId) {
                const player = this.players.get(slot.assignedEntityId);
                if (player) {
                    // Формула: Урон Врага - (Комбо защиты * Множитель)
                    // Для простоты: если слот пуст (count 0) -> урон
                    let dmg = 0;
                    if (slot.count === 0) dmg = 35;
                    else if (slot.count === 1) dmg = 15;
                    // X2+ блокирует урон полностью

                    if (dmg > 0) {
                        player.takeDamage(dmg);
                        console.log(`> DEFENSE FAIL D${slot.index+1}: ${player.name} took -${dmg} HP`);
                    } else {
                        console.log(`> BLOCKED D${slot.index+1}: ${player.name} is safe.`);
                    }
                }
            }
        });

        // 3. Охлаждение Latency
        this.players.forEach(p => p.ventLatency());

        // Проверка поражения
        const aliveCount = Array.from(this.players.values()).filter(p => p.status !== 'TERMINATED').length;
        if (aliveCount === 0) {
            console.log("MISSION FAILED. CONNECTION LOST.");
            this.isGameRunning = false;
            return;
        }

        this.startTurn();
    }
}
